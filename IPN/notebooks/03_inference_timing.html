3. Checkpoint Inspection¶

Load each checkpoint's metadata without building the full model. The conv1.weight tensor shape reveals the exact input-channel count and temporal kernel size that was used during training — this drives the model-building logic below.

In [3]:
def inspect_checkpoint(path: Path) -> dict:
    """Return architecture metadata extracted from a saved DataParallel checkpoint."""
    path = Path(path)
    if not path.exists():
        return {"file": path.name, "error": "FILE NOT FOUND"}

    ckpt = torch.load(path, map_location="cpu", weights_only=False)
    sd   = ckpt.get("state_dict", {})

    # First Conv3d (keys have 'module.' prefix from DataParallel)
    conv1_key = next(
        (k for k in sd if k.endswith("conv1.weight")), None
    )
    conv1_shape = tuple(sd[conv1_key].shape) if conv1_key else None
    # shape -> (out_ch, in_ch, kT, kH, kW)
    in_channels    = conv1_shape[1] if conv1_shape else "?"
    temporal_kern  = conv1_shape[2] if conv1_shape else "?"

    # Classification head
    fc_key = next(
        (k for k in sd if k.endswith("fc.weight") or k.endswith("classifier.1.weight")),
        None,
    )
    n_classes = int(sd[fc_key].shape[0]) if fc_key else "?"

    # Parameter count (includes all tensors in state_dict)
    n_params_M = sum(v.numel() for v in sd.values()) / 1e6

    return {
        "file"         : path.name,
        "arch"         : ckpt.get("arch", "unknown"),
        "epoch"        : ckpt.get("epoch", "?"),
        "in_channels"  : in_channels,
        "temporal_kern": temporal_kern,
        "conv1_shape"  : str(conv1_shape),
        "n_classes"    : n_classes,
        "n_params_M"   : round(n_params_M, 1),
    }

def _build_base_model(cfg: dict) -> nn.Module:
    """Instantiate the raw 3D-CNN (3-channel RGB, original kernel sizes)."""
    builder  = cfg["builder"]
    depth    = cfg["depth"]
    n_cls    = cfg["n_classes"]
    T        = cfg["sample_duration"]
    shortcut = cfg["shortcut_type"]
    S        = SAMPLE_SIZE

    if builder == "resnetl":
        from models.resnetl import resnetl10
        return resnetl10(
            num_classes=n_cls, shortcut_type=shortcut,
            sample_size=S, sample_duration=T
        )
    elif builder == "resnext":
        from models.resnext import resnet101
        return resnet101(
            num_classes=n_cls, shortcut_type=shortcut,
            cardinality=cfg.get("cardinality", 32),
            sample_size=S, sample_duration=T
        )
    elif builder == "resnet":
        from models.resnet import resnet50
        return resnet50(
            num_classes=n_cls, shortcut_type=shortcut,
            sample_size=S, sample_duration=T
        )
    else:
        raise ValueError(f"Unknown builder: {builder!r}")


def build_and_load(cfg: dict, device=DEVICE) -> nn.Module:
    """
    Build the model, apply the training-time channel modification inferred from
    the checkpoint's conv1.weight shape, then load the state dict.

    Training logic replicated from model.py:
      - in_channels == 4, temporal_kern == 3  ->  _construct_rgbdepth_model(model) + _modify_first_conv_layer(model, 3, 4)
      - in_channels == 4, temporal_kern == 7  ->  _construct_rgbdepth_model(model)
      - in_channels == 1                      ->  _construct_depth_model(model)
      - in_channels == 3, temporal_kern == 3  ->  _modify_first_conv_layer(model, 3, 3)
      - in_channels == 3, temporal_kern == 7  ->  base model (no modification)
    """
    ckpt_path = Path(cfg["ckpt"])
    ckpt = torch.load(ckpt_path, map_location="cpu", weights_only=False)
    sd   = ckpt["state_dict"]

    # -- Infer which channel modification was applied --------------------------
    conv1_key   = next(k for k in sd if k.endswith("conv1.weight"))
    conv1_shape = sd[conv1_key].shape  # (out_ch, in_ch, kT, kH, kW)
    in_ch       = conv1_shape[1]
    kT          = conv1_shape[2]

    model = _build_base_model(cfg)

    if in_ch == 4:
        # RGB-flo / RGB-seg: first conv was expanded from 3 -> 4 channels
        model = _construct_rgbdepth_model(model)
        if kT == 3:
            # no_first_lay path also shrinks temporal kernel 7 -> 3
            model = _modify_first_conv_layer(model, 3, 4)
    elif in_ch == 1:
        # Depth / seg-only stream
        model = _construct_depth_model(model)
    elif in_ch == 3 and kT == 3:
        # Plain RGB, first conv temporal kernel shrunk 7 -> 3
        model = _modify_first_conv_layer(model, 3, 3)
    # else: in_ch==3, kT==7 -> base model unchanged

    # -- Load weights (strip DataParallel 'module.' prefix) -------------------
    clean_sd = OrderedDict(
        (k[7:] if k.startswith("module.") else k, v) for k, v in sd.items()
    )
    model.load_state_dict(clean_sd, strict=True)
    model.eval()
    model.to(device)
    return model


def get_in_channels(cfg: dict) -> int:
    modality = cfg.get("modality", "RGB")
    return 4 if modality in ("RGB-D", "RGB-flo", "RGB-seg") else (1 if modality == "Depth" else 3)

5. Individual Model Benchmarks (FP32)¶

Each model is:

  1. Built and loaded from its checkpoint
  2. Warmed up with N_WARMUP un-timed forward passes (fills GPU pipeline)
  3. Timed for N_RUNS passes using torch.cuda.synchronize() around each call

All timings are wall-clock milliseconds for a single clip (batch size = 1).

In [5]:
def benchmark(
    model: nn.Module,
    dummy: torch.Tensor,
    n_warmup: int = N_WARMUP,
    n_runs: int = N_RUNS,
) -> dict:
    """
    Returns per-run latencies (ms) and aggregate statistics.
    Uses CUDA synchronisation barriers for accurate GPU timing.
    """
    sync = torch.cuda.synchronize if DEVICE.type == "cuda" else (lambda: None)

    model.eval()
    with torch.no_grad():
        # -- Warmup -----------------------------------------------------------
        for _ in range(n_warmup):
            _ = model(dummy)
        sync()

        # -- Timed runs -------------------------------------------------------
        latencies_ms = []
        for _ in range(n_runs):
            sync()
            t0 = time.perf_counter()
            _ = model(dummy)
            sync()
            latencies_ms.append((time.perf_counter() - t0) * 1_000)

    lat = np.array(latencies_ms)
    return {
        "mean_ms" : float(np.mean(lat)),
        "std_ms"  : float(np.std(lat)),
        "min_ms"  : float(np.min(lat)),
        "p50_ms"  : float(np.percentile(lat, 50)),
        "p95_ms"  : float(np.percentile(lat, 95)),
        "p99_ms"  : float(np.percentile(lat, 99)),
        "max_ms"  : float(np.max(lat)),
        "fps"     : float(1_000.0 / np.mean(lat)),
        "raw"     : latencies_ms,
    }

6. Dual-Model Online Pipeline Benchmarks¶

The online inference system (online_test.py) runs two models sequentially per stride:

  1. Detector (ResNetL-10, 8-frame clip) — decides if a gesture is present
  2. Classifier (ResNeXt-101 or ResNet-50, 32-frame clip) — identifies the gesture

Both models share the same modality (RGB, RGB-flo, or RGB-seg). The pipeline latency determines the maximum real-time throughput achievable.

        if clf_name not in AVAILABLE_MODEL_REGISTRY:
            reason.append(f"missing classifier checkpoint: {clf_name}")
        skipped_pipelines[pipe_name] = "; ".join(reason)
        continue

    print(f"\nPipeline: {pipe_name}")
    print(f"  Detector   : {det_name}")
    print(f"  Classifier : {clf_name}")

    det_model = build_and_load(AVAILABLE_MODEL_REGISTRY[det_name])
    clf_model = build_and_load(AVAILABLE_MODEL_REGISTRY[clf_name])

    det_input = make_dummy_input(AVAILABLE_MODEL_REGISTRY[det_name])
    clf_input = make_dummy_input(AVAILABLE_MODEL_REGISTRY[clf_name])

    with torch.no_grad():
        # Warmup both models together
        for _ in range(N_WARMUP):
            _ = det_model(det_input)
            _ = clf_model(clf_input)
        sync()

        # Timed pipeline
        latencies_ms = []
        for _ in range(N_RUNS):
            sync()
            t0 = time.perf_counter()
            _ = det_model(det_input)
            _ = clf_model(clf_input)
            sync()
            latencies_ms.append((time.perf_counter() - t0) * 1_000)

    lat = np.array(latencies_ms)
    pipe_results[pipe_name] = {
        "det"      : det_name,
        "clf"      : clf_name,
        "det_ms"   : ind_results[det_name]["mean_ms"],
        "clf_ms"   : ind_results[clf_name]["mean_ms"],
        "mean_ms"  : float(np.mean(lat)),
        "std_ms"   : float(np.std(lat)),
        "p50_ms"   : float(np.percentile(lat, 50)),
        "p95_ms"   : float(np.percentile(lat, 95)),
        "p99_ms"   : float(np.percentile(lat, 99)),
        "fps"      : float(1_000.0 / np.mean(lat)),
        "raw"      : latencies_ms,
    }

    print(
        f"  Pipeline mean: {pipe_results[pipe_name]['mean_ms']:.2f} ms  "
        f"P95: {pipe_results[pipe_name]['p95_ms']:.2f} ms  "
        f"FPS: {pipe_results[pipe_name]['fps']:.1f}"
    )

    del det_model, clf_model
    if DEVICE.type == "cuda":
        torch.cuda.empty_cache()

if skipped_pipelines:
    print("\nSkipped pipelines:")
    for pipe_name, reason in skipped_pipelines.items():
        print(f"  - {pipe_name}: {reason}")

if not pipe_results:
    print("\nNo complete detector+classifier pipeline is available with current checkpoints.")

print("\nPipeline benchmarks complete.")

def _bench(model, dummy, n_warmup=N_WARMUP, n_runs=N_RUNS):
    sync = torch.cuda.synchronize if DEVICE.type == "cuda" else lambda: None
    with torch.no_grad():
        for _ in range(n_warmup):
            model(dummy)
        sync()
        t0 = torch.cuda.Event(enable_timing=True)
        t1 = torch.cuda.Event(enable_timing=True)
        lats = []
        for _ in range(n_runs):
            sync()
            if DEVICE.type == "cuda":
                t0.record(); model(dummy); t1.record()
                sync(); lats.append(t0.elapsed_time(t1))
            else:
                import time
                s = time.perf_counter(); model(dummy)
                lats.append((time.perf_counter()-s)*1000)
    return lats

for spec in TARGET_ARCHS:
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        model = spec.build_fn().to(DEVICE).eval()

    params_m = round(sum(p.numel() for p in model.parameters()) / 1e6, 1)
    dummy32  = torch.randn(BATCH_SIZE, spec.in_ch, spec.T, 112, 112, device=DEVICE)

    lats32 = _bench(model, dummy32)
    mean32 = round(sum(lats32) / len(lats32), 2)
    fps32  = round(1000 / mean32, 1)

    fp16_ok = DEVICE.type == "cuda"
    if fp16_ok:
        try:
            model16  = model.half()
            dummy16  = dummy32.half()
            lats16   = _bench(model16, dummy16)
            mean16   = round(sum(lats16) / len(lats16), 2)
            fps16    = round(1000 / mean16, 1)
            speedup  = f"{mean32/mean16:.2f}x"
        except Exception as e:
            mean16 = fps16 = float("nan"); speedup = "ERR"
    else:
        mean16 = fps16 = float("nan"); speedup = "N/A"

    print(f" {spec.label:<25} {spec.T:>6}  {params_m:>9}  {mean32:>8.2f}  {fps32:>9.1f}  "
          f"{mean16!s:>8}  {fps16!s:>9}  {speedup:>7}")
